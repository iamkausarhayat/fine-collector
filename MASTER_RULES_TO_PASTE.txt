1. CLOUD FIRESTORE RULES (Paste into Firebase Console > Firestore Database > Rules)
--------------------------------------------------------------------------------
Copy everything between the dashed lines and paste into Firestore Rules, then click "Publish":

------------------------------------[COPY START]------------------------------------
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() { return request.auth != null && exists(/databases/$(database)/documents/admins/$(request.auth.uid)); }
    function isMaster() { return isAdmin() && get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.role == 'master'; }
    function validStudent(d) {
      return d.keys().hasOnly(['id','name','fine','date','time','paid','createdAt','updatedAt'])
        && d.keys().hasAll(['id','name','fine','date','time','paid','createdAt'])
        && d.name is string && d.name.size() > 0 && d.name.size() <= 60
        && d.fine is number && d.fine >= 0 && d.fine <= 1000
        && d.date is string && d.date.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')
        && d.time is string && d.time.size() <= 20
        && d.paid is bool;
    }
    match /students/{id} {
      allow read: if true;
      allow create, update: if isAdmin() && validStudent(request.resource.data);
      allow delete: if isAdmin();
    }
    match /admins/{uid} {
      allow read: if request.auth != null && (request.auth.uid == uid || isMaster());
      allow write: if isMaster();
    }
    match /admin_requests/{uid} {
      allow create: if request.auth != null && request.auth.uid == uid
        && request.resource.data.keys().hasOnly(['uid','name','device','requestedAt','status'])
        && request.resource.data.status == 'pending'
        && request.resource.data.name is string && request.resource.data.name.size() <= 60;
      allow read: if request.auth != null && (request.auth.uid == uid || isMaster());
      allow update, delete: if isMaster();
    }
    match /admin_audit_logs/{id} {
      allow read: if isMaster();
      allow create: if isAdmin() && request.resource.data.adminUid == request.auth.uid;
      allow update, delete: if false;
    }
    match /{document=**} { allow read, write: if false; }
  }
}
